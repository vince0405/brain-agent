Use Brain in Your Production App

Drop-in Python SDK (sdk/brain_client.py) or standard HTTP/WebSocket calls:

from sdk.brain_client import BrainClient

brain = BrainClient(base_url="http://localhost:8000", api_key="your-key")

# Full agent turn with memory + RAG + tool calling
res = brain.chat(
    message="Summarize our SLA and calculate 99.95% uptime in minutes/month",
    user_id="customer_42",
    session_id="ticket_901",
)
print(res["reply"])
print("Tools used:", [t["tool_name"] for t in res["tool_calls"]])
curl -X POST http://localhost:8000/api/v1/chat \
  -H "Content-Type: application/json" \
  -H "X-API-Key: your-production-key" \
  -d '{
    "message": "What do you remember about my stack?",
    "user_id": "default_user",
    "enable_tools": true,
    "enable_rag": true,
    "enable_memory": true
  }'
